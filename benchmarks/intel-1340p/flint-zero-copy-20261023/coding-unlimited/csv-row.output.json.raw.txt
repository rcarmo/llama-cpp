function parseCsvRow(row) {
  const input = typeof row === "string" ? row : "";
  if (input.includes("\r") || input.includes("\n")) {
    throw Error("Invalid row");
  }

  const cells = [];
  let cell = "";
  let inQuotes = false;
  let quoted = false;
  let closed = false;
  let i = 0;

  while (i < input.length) {
    const ch = input[i];

    if (inQuotes) {
      if (ch === '"') {
        if (input[i + 1] === '"') {
          cell += '"';
          i += 2;
        } else {
          inQuotes = false;
          closed = true;
          i += 1;
          if (i < input.length && input[i] !== ',') {
            throw Error("Invalid character after closing quote");
          }
        }
      } else {
        cell += ch;
        i += 1;
      }
    } else {
      if (ch === ',') {
        cells.push(cell);
        cell = "";
        quoted = false;
        closed = false;
        i += 1;
      } else if (ch === '"') {
        if (cell === "" && !quoted && !closed) {
          inQuotes = true;
          quoted = true;
          i += 1;
        } else {
          throw Error("Invalid quote");
        }
      } else {
        if (closed) {
          throw Error("Invalid character after closing quote");
        }
        cell += ch;
        i += 1;
      }
    }
  }

  if (inQuotes) {
    throw Error("Unclosed quote");
  }

  cells.push(cell);
  return cells;
}

module.exports = parseCsvRow;
